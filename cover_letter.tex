%!TEX TS-program = xelatex
%!TEX encoding = UTF-8 Unicode
% Awesome CV LaTeX Template for Cover Letter
%
% The template for this document is by:
% Claud D. Park <posquit0.bj@gmail.com>
% https://github.com/posquit0/Awesome-CV
%
% Author:
% William H. Weiskopf <william@weiskopf.me>
%
% This is a generic template: it holds no company-specific content. A specific
% letter lives outside this (public) repo, defines any of the \Letter... macros
% below, then \input{cover_letter.tex}, compiling with this repo on the TeX search path
% (e.g. `tectonic -Z search-path=<this repo> letter.tex`). Undefined macros fall back to the
% placeholders here, so this file also compiles on its own as a sample.
%

%-------------------------------------------------------------------------------
% LETTER CONTENT (defaults; override before \input-ing this file)
%-------------------------------------------------------------------------------
\providecommand{\LetterRecipientName}{Hiring Team}
\providecommand{\LetterRecipientAddress}{Company\\City, State}
\providecommand{\LetterDate}{\today}
\providecommand{\LetterTitle}{Application for Staff Software Engineer}
\providecommand{\LetterOpening}{Dear Hiring Team,}
\providecommand{\LetterClosing}{Sincerely,}
\providecommand{\LetterBody}{%
  [Why this company: what draws me to its mission and problems, specifically.]

  [Why me: one or two stories that match the role.]

  [Close: what I'd like to talk about next.]
}

%-------------------------------------------------------------------------------
% CONFIGURATIONS
%-------------------------------------------------------------------------------
\documentclass[11pt, letterpaper]{awesome-cv}

% Configure page margins with geometry
\geometry{left=1.4cm, top=.8cm, right=1.4cm, bottom=1.8cm, footskip=.5cm}

% Specify the location of the included fonts
\fontdir[fonts/]

% Color for highlights
\colorlet{awesome}{awesome-red}

% Set false if you don't want to highlight section with awesome color
\setbool{acvSectionColorHighlight}{true}

% If you would like to change the social information separator from a pipe (|) to something else
\renewcommand{\acvHeaderSocialSep}{\quad\textbar\quad}

\input{header.tex}

%-------------------------------------------------------------------------------
%  LETTER INFORMATION
%-------------------------------------------------------------------------------
\recipient{\LetterRecipientName}{\LetterRecipientAddress}
\letterdate{\LetterDate}
\lettertitle{\LetterTitle}
\letteropening{\LetterOpening}
\letterclosing{\LetterClosing}
\letterenclosure[Attached]{Résumé}

%-------------------------------------------------------------------------------
\begin{document}

% Print the header with above personal informations
\makecvheader

% Print the footer with 3 arguments(<left>, <center>, <right>)
\makecvfooter
{\today}
{}
{}

% Print the title with above letter informations
\makelettertitle

%-------------------------------------------------------------------------------
%  LETTER CONTENT
%-------------------------------------------------------------------------------
\begin{cvletter}
  \LetterBody
\end{cvletter}

%-------------------------------------------------------------------------------
% Print the signature and enclosures with above letter informations
\makeletterclosing

\end{document}
